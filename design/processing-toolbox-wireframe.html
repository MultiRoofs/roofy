CREATE OR REPLACE TEMP TABLE solid_scratch AS
WITH parsed AS (
  SELECT id, feature_id, ST_3DTryFromWKB(wkb) AS g FROM solid_src
)
SELECT id, feature_id,
       ST_3DValidationReport(g).is_valid       AS solid_valid,
       CASE WHEN ST_3DValidationReport(g).is_valid
            THEN ST_3DVolume(g) END            AS solid_volume_m3,
       ST_3DArea(g)                            AS solid_envelope_m2,
       ST_3DFootprintArea(g)                   AS solid_footprint_m2,
       ST_3DZMin(g)                            AS zmin_m,
       ST_3DZMax(g)                            AS zmax_m
FROM parsed
WHERE g IS NOT NULL;
0.771 s1,116 rows · 37 invalid
4 · Computing · roll-up to features 
CREATE OR REPLACE TEMP TABLE solid_rollup AS
SELECT feature_id,
       CASE WHEN bool_and(solid_volume_m3 IS NOT NULL)
            THEN sum(solid_volume_m3) END      AS solid_volume_m3,
       sum(solid_envelope_m2)                  AS solid_envelope_m2,
       sum(solid_footprint_m2)                 AS solid_footprint_m2,
       max(zmax_m) - min(zmin_m)               AS solid_height_m,
       bool_and(solid_valid)                   AS solid_valid
FROM solid_scratch
GROUP BY feature_id;
